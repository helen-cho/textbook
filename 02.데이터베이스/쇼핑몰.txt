-- 상품 테이블 생성
CREATE TABLE PRODUCT (
    PROD_ID    NUMBER(10)          PRIMARY KEY, -- 상품번호
    PROD_NAME  VARCHAR2(100)       NOT NULL,    -- 상품명
    CAT_ID     VARCHAR2(20),                    -- 카테고리ID
    PRICE      NUMBER(12)          DEFAULT 0,   -- 가격
    STOCK      NUMBER(10)          DEFAULT 0,   -- 재고수량
    REG_DATE   DATE                DEFAULT SYSDATE, -- 등록일,
    FOREIGN KEY(CAT_ID) REFERENCES CATEGORY(CAT_ID)
);

--주문 기본정보 테이블 생성
CREATE TABLE ORDERS (
    ORDER_ID    NUMBER(10)      PRIMARY KEY,   -- 주문번호
    USER_ID     VARCHAR2(20)    NOT NULL,      -- 주문자ID (FK)
    ORDER_DATE  DATE            DEFAULT SYSDATE, -- 주문일자
    TOTAL_PRICE NUMBER(12)      DEFAULT 0,     -- 총 결제금액
    STATUS      VARCHAR2(20)    DEFAULT 0, -- 주문상태
    FOREIGN KEY(USER_ID) REFERENCES MEMBER(USER_ID)
);

    DETAIL_ID   NUMBER(10)      PRIMARY KEY,   -- 상세번호
    ORDER_ID    NUMBER(10)      NOT NULL,      -- 주문번호 (FK)
    PROD_ID     NUMBER(10)      NOT NULL,      -- 상품번호 (FK)

);
